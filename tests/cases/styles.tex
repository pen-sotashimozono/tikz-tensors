% A specimen sheet, not a diagram.
%
% Every public style and command of the package appears here once, each on its
% own with the name it is there to show written under it. tests/coverage.py
% fails if a name in tex/ is used by no file under tests/cases/ or examples/, so
% a new style arrives with its specimen here; and this is the appearance test's
% main case, so a change to any of them shows up as a changed reference image.
%
% Specimens, deliberately: a sheet that is laid out as a picture starts to look
% like it is claiming something, and this one is not. The one place several
% names are shown together is the chain, because a triangle means nothing on its
% own -- it says "isometric from this side", which needs the other side and the
% centre to be visible. Everything else stands alone.
\documentclass[border=5pt]{standalone}
\usepackage{amsmath}
\usepackage{tikz-tensors}

% a caption under a specimen, naming what it is a specimen of
\newcommand{\spec}[3]{\node[font=\tiny\ttfamily, text=ttmuted, anchor=north,
                            align=center] at (#1,#2) {#3};}

\begin{document}
\begin{tikzpicture}
  % ---- nodes -----------------------------------------------------------
  \node[fn]       (f)  at (0,0)    {$\varphi$};   \spec{0}{-0.85}{fn}
  \node[fnwide]   (fw) at (1.7,0)  {$\psi$};      \spec{1.7}{-0.85}{fnwide}
  \node[fntall]   (ft) at (3.3,0)  {$\chi$};      \spec{3.3}{-1.2}{fntall}
  \node[coef]     (c)  at (4.5,0)  {$c$};         \spec{4.5}{-0.85}{coef}
  \node[coefwide] (cw) at (6.0,0)  {$C$};         \spec{6.0}{-0.85}{coefwide}
  \node[coeftall] (ct) at (7.5,0)  {$A$};         \spec{7.5}{-1.2}{coeftall}
  \node[op]       (o)  at (8.6,0)  {$\hat H$};    \spec{8.6}{-0.85}{op}
  \node[opwide]   (ow) at (10.1,0) {$W$};         \spec{10.1}{-0.85}{opwide}
  \node[coef, tn round] (cr) at (11.5,0) {$A$};   \spec{11.5}{-0.85}{coef,\\tn round}

  % ---- edges and what goes on them -------------------------------------
  \tnbond[cont]{(0,-2.2) -- (0.9,-2.2)}                  \spec{0.45}{-2.4}{cont}
  \tnbond[disc]{(1.7,-2.2) -- (2.6,-2.2)}                \spec{2.15}{-2.4}{disc}
  \tnbond{(3.4,-2.2) -- (4.3,-2.2)}                      \spec{3.85}{-2.4}{bond}
  \begin{scope}[tn arrows=false]
    \tnbond{(5.1,-2.2) -- (6.0,-2.2)}                    \spec{5.55}{-2.4}{tn arrows=false}
  \end{scope}
  \tnbond[disc]{(7.0,-2.2) -- (7.9,-2.2) node[leg, right] {$i$}}
                                                         \spec{7.45}{-2.4}{leg}
  \draw[tn line, draw=exact] (9.0,-2.15) -- (9.9,-2.15);
  \begin{scope}[tn line width=1.8pt]
    \draw[tn line, draw=exact] (9.0,-2.45) -- (9.9,-2.45);
  \end{scope}                                            \spec{9.45}{-2.7}{tn line,\\tn line width}
  \node[coef] (q1) at (11.1,-2.2) {}; \node[coef] (q2) at (11.9,-2.2) {};
  \draw[frame] ([shift={(-0.15,0.15)}]q1.north west)
       rectangle ([shift={(0.15,-0.15)}]q2.south east); \spec{11.5}{-2.65}{frame}

  % ---- exchanging two legs, both kinds ---------------------------------
  \tnswap{(0.1,-3.8)}{(0.9,-3.8)}{0.9}                   \spec{0.5}{-4.9}{\textbackslash tnswap}
  \tnswap[disc]{(2.1,-3.8)}{(2.9,-3.8)}{0.9}             \spec{2.5}{-4.9}{\textbackslash tnswap[disc]}

  % ---- the one group that has to be shown together ---------------------
  % canl and canr say "isometric from the left" and "from the right", which is
  % a statement about a chain; centre is the one tensor that is neither. Alone
  % they would be three shapes with nothing to be read against.
  \node[canl]   (A1) at (4.6,-4.0) {};
  \node[canl]   (A2) at (5.9,-4.0) {};
  \node[centre] (C)  at (7.2,-4.0) {};
  \node[canr]   (B1) at (8.5,-4.0) {};
  \node[canr]   (B2) at (9.8,-4.0) {};
  \tnbond{(A1) -- (A2)}  \tnbond{(A2) -- (C)}
  \tnbond{(B1) -- (C)}   \tnbond{(B2) -- (B1)}
  \tnbond{([xshift=-5mm]A1.west) -- (A1)}
  \tnbond{([xshift=5mm]B2.east) -- (B2)}
  \tngate{G}{$U$}{A2,C}{-5.3}
  \tngatelegs{G}{north}{l}{A2}
  \tngatelegs{G}{north}{c}{C}
  % An operator on the chain. Its bond and the legs into it are drawn before its
  % tensors here and would be drawn across them without the edge layer; the
  % \node calls come last on purpose, and the lines still go under.
  \tnbond[disc]{(A1.corner 3 |- 0,-6.6) -- (B2.corner 2 |- 0,-6.6)}
  \tnlegs{l}{-6.6}{A1}  \tnlegs{r}{-6.6}{B1,B2}
  \tnbond[disc]{(G.south -| A2.corner 3) -- (A2.corner 3 |- 0,-6.6)}
  \tnbond[disc]{(G.south -| C) -- (C |- 0,-6.6)}
  \foreach \p in {{A1.corner 3},{A2.corner 3},{C},{B1.corner 2},{B2.corner 2}}
    \tnbond[disc]{(\p |- 0,-6.6) -- (\p |- 0,-7.4)};
  \foreach \p in {{A1.corner 3},{A2.corner 3},{C},{B1.corner 2},{B2.corner 2}}
    \node[mpo] at (\p |- 0,-6.6) {};
  \spec{7.2}{-7.6}{canl, canr, centre, mpo, gate, bond\\%
    \textbackslash tnbond, \textbackslash tnlegs, \textbackslash tngate,
    \textbackslash tngatelegs}

  % ---- the two that are easier to see on their own ---------------------
  \node[canl]       (Y1) at (0.4,-6.4) {};
  \node[centrebond] (Yl) at (1.4,-6.4) {};
  \node[canr]       (Y2) at (2.4,-6.4) {};
  \tnbond{(Y1) -- (Yl)}  \tnbond{(Y2) -- (Yl)}           \spec{1.4}{-6.95}{centrebond}
  \node[canl] (V1) at (0.9,-8.3) {};
  \tngate{V}{$V$}{V1}{-9.2}
  \tngatelegs{V}{north}{l}{V1}                           \spec{0.9}{-9.6}{\textbackslash tngate,
                                                             one site}

  % ---- the named colours -----------------------------------------------
  % `color=` and not a bare `\fill[\c]`, because several of these names are also
  % styles (coef, mpo, gate, centre), and TikZ resolves an option as a key before
  % it tries it as a colour -- a bare one silently fills with the style instead.
  \foreach \c [count=\i] in {ele,nuc,coef,exchange,exact,%
                             canl,canr,centre,mpo,gate,%
                             ttfg,ttmuted,ttaccent,ttgood,ttwarn,ttbad}
    \fill[color=\c] (3.6+\i*0.6-0.6,-8.6) rectangle ++(0.45,0.3);
  \spec{7.8}{-8.9}{the named colours}
\end{tikzpicture}
\end{document}
