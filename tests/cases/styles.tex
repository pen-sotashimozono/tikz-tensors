% Every public style and command once, side by side: the appearance test's
% main case. tests/coverage.py fails if a style in tex/ is used nowhere under
% tests/cases/, so a new style arrives with its picture here.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}
  % nodes: functions, arrays, operators
  \node[fn]       (f)  at (0,0)    {$\varphi$};
  \node[fnwide]   (fw) at (1.9,0)  {$\psi$};
  \node[fntall]   (ft) at (3.7,0)  {$\chi$};
  \node[coef]     (c)  at (5.0,0)  {$c$};
  \node[coefwide] (cw) at (6.7,0)  {$C$};
  \node[coeftall] (ct) at (8.4,0)  {$A$};
  \node[op]       (o)  at (9.6,0)  {$\hat H$};
  \node[opwide]   (ow) at (11.3,0) {$W$};
  % legs: continuous and discrete, with labels
  \draw[cont] (f.south) -- ++(0,-1) node[leg, below] {$\mathbf r$};
  \draw[disc] (c.south) -- ++(0,-1) node[leg, below] {$i$};
  \draw[disc] (f.east) -- (fw.west);
  % a group contracting to one object
  \draw[frame] ([shift={(-0.15,0.15)}]c.north west) rectangle ([shift={(0.15,-0.15)}]cw.south east);
  % exchange of two legs, both kinds
  \tnswap{([xshift=-4.5mm]fw.south)}{([xshift=4.5mm]fw.south)}{1}
  \tnswap[disc]{([xshift=-4.5mm]ow.south)}{([xshift=4.5mm]ow.south)}{1}
  % Matrix-product states, stacked in the order the parts are applied and read
  % downward: the chain in mixed canonical form; a two-site gate on the physical
  % legs of two of its sites; then an operator on the chain, one tensor per site,
  % which takes what comes out of the gate and the three legs that passed the
  % gate by. Every leg crosses a clear gap, so nothing here is a tensor sitting
  % on another tensor.
  \node[env]    (EL) at (-1.1,-4.2) {};
  \node[canl]   (A1) at (0,-4.2)    {};
  \node[canl]   (A2) at (1.3,-4.2)  {};
  \node[centre] (C)  at (2.6,-4.2)  {};
  \node[canr]   (B1) at (3.9,-4.2)  {};
  \node[canr]   (B2) at (5.2,-4.2)  {};
  \node[envc]   (ER) at (6.3,-4.2)  {};
  \draw[bond] (EL) -- (A1);  \draw[bond] (A1) -- (A2);  \draw[bond] (A2) -- (C);
  \draw[bond] (B1) -- (C);   \draw[bond] (B2) -- (B1);  \draw[bond] (ER) -- (B2);
  \node[gate, minimum width=24mm] (G) at (1.95,-5.5) {$U$};
  \tngatelegs{G}{north}{A2,C}
  % the operator's own bond first, so that the tensors cover it
  \draw[disc] (A1 |- 0,-6.8) -- (B2 |- 0,-6.8);
  \foreach \n in {A1,A2,C,B1,B2} \node[mpo] (W\n) at (\n |- 0,-6.8) {};
  \foreach \n in {A1,B1,B2} \draw[disc] (\n) -- (W\n);
  \foreach \n in {A2,C}    \draw[disc] (G.south -| \n) -- (W\n);
  \foreach \n in {A1,A2,C,B1,B2} \draw[disc] (W\n.south) -- ++(0,-0.6);

  % The centre on a bond -- the Schmidt values, no physical leg -- and the one
  % stroke everything is drawn with: `tn line` on a path of this file's own, and
  % `tn line width` on a scope, which moves outlines and legs together.
  \begin{scope}[xshift=7.6cm, yshift=-4.2cm, tn line width=1.6pt]
    \node[canl]       (Z1) at (0,0)    {};
    \node[centrebond] (Zl) at (1.15,0) {};
    \node[canr]       (Z2) at (2.3,0)  {};
    \draw[bond] (Z1) -- (Zl);  \draw[bond] (Z2) -- (Zl);
    \draw[tn line, draw=exact] (0,-0.9) -- (2.3,-0.9);
  \end{scope}

  % The named colours: the physics ones, the MPS ones, then the page tokens.
  % `color=` and not a bare `\fill[\c]`, because several of these names are also
  % styles (coef, mpo, gate, centre), and TikZ resolves an option as a key before
  % it tries it as a colour -- a bare one silently fills with the style instead.
  \foreach \c [count=\i] in {ele,nuc,coef,exchange,exact,%
                              canl,canr,centre,mpo,gate,edge,edgec,%
                              ttfg,ttmuted,ttaccent,ttwarn,ttbad}
    \fill[color=\c] (\i*0.6-0.6,-2.1) rectangle ++(0.45,0.3);
\end{tikzpicture}
\end{document}
