% A specimen sheet, not a diagram.
%
% Every public style and command of the package appears here once, each on its
% own with the name it is there to show written under it. tests/coverage.py
% fails if a name in tex/ is used by no file under tests/cases/ or examples/, so
% a new style arrives with its specimen here; and this is the appearance test's
% main case, so a change to any of them shows up as a changed reference image.
%
% The package's names are shapes and lines, and so is this sheet: it loads no
% notation, and nothing on it stands for anything. What a triangle or a wavy
% leg means is a convention's to say (examples/conventions/notation.tex is the
% one the examples use), and a sheet of the package should not borrow one.
%
% Specimens, deliberately: a sheet that is laid out as a picture starts to look
% like it is claiming something, and this one is not. The few groups are there
% because a command lays out several things at once.
\documentclass[border=5pt]{standalone}
\usepackage{amsmath}
\usepackage{tikz-tensors}

% a caption under a specimen, naming what it is a specimen of
\newcommand{\spec}[3]{\node[font=\tiny\ttfamily, text=ttmuted, anchor=north,
                            align=center] at (#1,#2) {#3};}

% the tokens are the package's to set and a notation's to change; set here to
% what they already are, so that the sheet shows \tnset without moving anything
\tnset{stub=4mm, line width=0.8pt}

\begin{document}
\begin{tikzpicture}
  % ---- shapes ----------------------------------------------------------
  \node[tn node]          at (0,0)    {$T$};  \spec{0}{-0.85}{tn node}
  \node[tn box]           at (1.3,0)  {$T$};  \spec{1.3}{-0.85}{tn box}
  \node[tn circle]        at (2.6,0)  {$T$};  \spec{2.6}{-0.85}{tn circle}
  \node[tn rounded]       at (3.9,0)  {$T$};  \spec{3.9}{-0.85}{tn rounded}
  \node[tn triangle right] at (5.3,0) {$T$};  \spec{5.3}{-0.85}{tn triangle\\right}
  \node[tn triangle left] at (6.8,0)  {$T$};  \spec{6.8}{-0.85}{tn triangle\\left}
  \node[tn diamond]       at (8.2,0)  {$T$};  \spec{8.2}{-0.85}{tn diamond}
  \node[tn diamond, tn small] at (9.4,0) {$T$}; \spec{9.4}{-0.85}{tn diamond,\\tn small}
  \node[tn box, tn fill=blue1] at (10.6,0) {$T$}; \spec{10.6}{-0.85}{tn box,\\tn fill}

  \node[tn capsule, tn wide] at (2.9,-2.0) {$T$}; \spec{2.9}{-2.85}{tn capsule,\\tn wide}
  \node[tn box, tn tall]     at (4.6,-2.0) {$T$}; \spec{4.6}{-3.2}{tn box,\\tn tall}
  \node[tn capsule, tn tall] at (5.8,-2.0) {$T$}; \spec{5.8}{-3.2}{tn capsule,\\tn tall}
  \node[tn gate]             at (7.1,-2.0) {$U$}; \spec{7.1}{-2.85}{tn gate}

  % ---- the named colours -----------------------------------------------
  % `color=` and not a bare `\fill[\c]`, because a convention may give a style
  % the name of a colour, and TikZ resolves an option as a key before it tries
  % it as a colour -- a bare one would silently fill with the style instead.
  \foreach \c [count=\i] in {ele,nuc,coef,exchange,exact,%
                             canl,canr,centre,mpo,gate,env,%
                             ttfg,ttmuted,ttaccent,ttgood,ttwarn,ttbad}
    \fill[color=\c] (0.4+\i*0.6-0.6,-6.0) rectangle ++(0.45,0.3);
  \spec{5.2}{-6.1}{the named colours}

  % ---- stacks, in rows of their own --------------------------------------
  % A stack places itself: \tnbreak starts a row under everything above, a
  % relation sign goes after what is on the row, and a stack after another
  % stands beside it. So these rows have no coordinates, and their captions
  % hang under whatever the row turned out to be.
  \tnbreak
  \tnstack[left=$L$, right=$R$]{X}{2}{a, b, c}
  \tnlayer{X}{a}{tn triangle right/$A$, .}
  \tnlayer{X}{b}{tn box/$T$, tn rounded/$W$}
  \tnlayer{X}{c}{., -}
  \tnconnect{X}
  \tnopen{up}{X-b-2}
  \tneq[factor=$\lambda$]
  \tnstack[left=$L$]{Y}{0}{a, b, c}
  \tnconnect{Y}
  % where a stack takes a tensor's index from: a triangle sets it to a corner
  % of its flat side; any shape can, and the stack moves the tensor across so
  % that point is on the site's line
  \tnstack{Z}{1}{a, b}
  \tnlayer{Z}{a}{{tn box, tn leg anchor=south west}/$T$}
  \tnlayer{Z}{b}{tn box/$T$}
  \tnconnect{Z}
  \node[font=\tiny\ttfamily, text=ttmuted, anchor=north, align=center]
    at (current bounding box.south) {\textbackslash tnstack, \textbackslash tnlayer,
    \textbackslash tnconnect, \textbackslash tnopen, \textbackslash tneq, tn leg anchor};

  % ---- a second row of stacks: what a grid can hold ---------------------
  % \tnbreak starts a row under everything above. One stack holds a tensor two
  % layers tall, a `dots' slot, a copy tensor, the slots a site's index runs
  % through, fused bonds, and an arrow taken from a shape that says which way
  % it points; the next, two indices exchanged on their way down.
  \tnbreak
  \tnstack{W}{5}{a, b, c}
  \tnlayer{W}{a}{tn box/$Z$/1/2, tn box, dots, tn dot/, {tn box, tn points=left}/$P$}
  \tnlayer{W}{b}{., tn rounded, |, |, tn box}
  \tnlayer{W}{c}{|, |, |, |, |}
  \tnconnect[along=tn arrow, apart=c]{W}
  \tnopen{down}{W}
  \tnstack{V}{2}{a, b}
  \tnlayer{V}{a}{tn box/$\Psi$/2}
  \tnlayer{V}{b}{|, |}
  \tnopenswap[edge=tn wavy, labels={$a$, $b$}]{V-a-1}
  \tnstack{U}{2}{a}
  \tnlayer{U}{a}{tn box, tn box}
  \tnconnect[along=tn double]{U}
  % a tree on the same grid: triangles pointing up and down, each as wide as
  % the sites under it, with one index on the apex side (tn single)
  \tnstack{R}{4}{a, b}
  \tnlayer{R}{a}{tn triangle down/$D$/4}
  \tnlayer{R}{b}{tn triangle up/$U$/2, {tn box, tn single=up}/$S$/2}
  \tnconnect[apart={a, b}]{R}
  \tnopen{up}{R-a-1}
  \tnopen{down}{R}
  \node[font=\tiny\ttfamily, text=ttmuted, anchor=north, align=center]
    at (current bounding box.south) {\textbackslash tnbreak, tn box/\$Z\$/1/2, dots,
    tn dot, |, apart=, tn points=,\\\textbackslash tnopenswap, tn double, tn triangle up, tn triangle down, tn single};

  % ---- a third row: a chain repeated, labels, and joins ------------------
  % <n>* repeats an entry; \tnjoin closes the chain into a ring, round the
  % top of the stack, and traces an operator in the gutter beside it.
  \tnbreak
  \tnstack{C}{5}{a}
  \tnlayer{C}{a}{2*tn triangle right/$A$, tn diamond/$C$, 2*tn triangle left/$B$}
  \tnconnect[along=tn arrow]{C}
  \tnopen[label=$i_{#1}$]{down}{C}
  \tnjoin{C-a-1:left}{C-a-5:right}
  \tnmid{C-a-1}{C-a-2}{$D$}
  \tnstack{J}{1}{a}
  \tnlayer{J}{a}{tn rounded/$O$}
  \tnjoin{J-a-1:up}{J-a-1:down}
  \node[font=\tiny\ttfamily, text=ttmuted, anchor=north, align=center]
    at (current bounding box.south) {2*, \textbackslash tnjoin, label=, labels=, edge=, \textbackslash tnput,
    \textbackslash tnmid, \textbackslash tnset};
