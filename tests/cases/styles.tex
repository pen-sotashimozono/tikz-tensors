% Every public style and command once, side by side: the appearance test's
% main case. tests/coverage.py fails if a style in tex/ is used nowhere under
% tests/cases/, so a new style arrives with its picture here.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}
  % nodes: functions, arrays, operators
  \node[fn]       (f)  at (0,0)    {$\varphi$};
  \node[fnwide]   (fw) at (1.9,0)  {$\psi$};
  \node[fntall]   (ft) at (3.7,0)  {$\chi$};
  \node[coef]     (c)  at (5.0,0)  {$c$};
  \node[coefwide] (cw) at (6.7,0)  {$C$};
  \node[coeftall] (ct) at (8.4,0)  {$A$};
  \node[op]       (o)  at (9.6,0)  {$\hat H$};
  \node[opwide]   (ow) at (11.3,0) {$W$};
  % legs: continuous and discrete, with labels
  \draw[cont] (f.south) -- ++(0,-1) node[leg, below] {$\mathbf r$};
  \draw[disc] (c.south) -- ++(0,-1) node[leg, below] {$i$};
  \draw[disc] (f.east) -- (fw.west);
  % a group contracting to one object
  \draw[frame] ([shift={(-0.15,0.15)}]c.north west) rectangle ([shift={(0.15,-0.15)}]cw.south east);
  % exchange of two legs, both kinds
  \tnswap{([xshift=-4.5mm]fw.south)}{([xshift=4.5mm]fw.south)}{1}
  \tnswap[disc]{([xshift=-4.5mm]ow.south)}{([xshift=4.5mm]ow.south)}{1}
  % Matrix-product states, stacked in the order the parts are applied and read
  % downward: a finite chain in mixed canonical form; a two-site gate on the
  % physical legs of two of its sites, sized and placed by \tngate from those
  % sites; then an operator on the chain, one tensor per site, which takes what
  % comes out of the gate and the three legs that passed it by. Every leg
  % crosses a clear gap, so nothing here is a tensor sitting on another tensor.
  %
  % The physical legs leave the triangles at the bottom corner of the flat side,
  % mirrored between the two gauges, which is \tnlegs{l} and \tnlegs{r}; the
  % diamond takes {c}, from its own border.
  \node[canl]   (A1) at (0,-4.2)    {};
  \node[canl]   (A2) at (1.3,-4.2)  {};
  \node[centre] (C)  at (2.6,-4.2)  {};
  \node[canr]   (B1) at (3.9,-4.2)  {};
  \node[canr]   (B2) at (5.2,-4.2)  {};
  \draw[bond] (A1) -- (A2);  \draw[bond] (A2) -- (C);
  \draw[bond] (B1) -- (C);   \draw[bond] (B2) -- (B1);
  \tngate{G}{$U$}{A2,C}{-5.5}
  \tngatelegs{G}{north}{l}{A2}
  \tngatelegs{G}{north}{c}{C}
  % the operator's own bond first, so that the tensors cover it
  \draw[disc] (A1.corner 3 |- 0,-6.8) -- (B2.corner 2 |- 0,-6.8);
  \foreach \p in {{A1.corner 3},{A2.corner 3},{C},{B1.corner 2},{B2.corner 2}}
    \node[mpo] at (\p |- 0,-6.8) {};
  \tnlegs{l}{-6.8}{A1}  \tnlegs{r}{-6.8}{B1,B2}
  \draw[disc] (G.south -| A2.corner 3) -- (A2.corner 3 |- 0,-6.8);
  \draw[disc] (G.south -| C) -- (C |- 0,-6.8);
  \foreach \p in {{A1.corner 3},{A2.corner 3},{C},{B1.corner 2},{B2.corner 2}}
    \draw[disc] (\p |- 0,-7.18) -- ++(0,-0.5);

  % A chain with the gauge left unstated -- `tn arrows=false' draws the same
  % bond without its arrowhead -- a one-site gate, the centre on a bond, and the
  % one stroke everything is drawn with: `tn line' on a path of this file's own,
  % `tn line width' on a scope, which moves outlines and legs together.
  \begin{scope}[xshift=7.8cm, yshift=-4.2cm, tn arrows=false]
    \node[canl]       (Y1) at (0,0)    {};
    \node[centrebond] (Yl) at (1.15,0) {};
    \node[canr]       (Y2) at (2.3,0)  {};
    \draw[bond] (Y1) -- (Yl);  \draw[bond] (Y2) -- (Yl);
    \tngate{V}{$V$}{Y1}{-1.4}
    \tngatelegs{V}{north}{l}{Y1}
    \tnlegs{r}{-1.4}{Y2}
  \end{scope}
  \begin{scope}[xshift=7.8cm, yshift=-6.6cm, tn line width=1.6pt]
    \node[canl] (Z1) at (0,0)   {};
    \node[canr] (Z2) at (1.8,0) {};
    \draw[bond] (Z1) -- (Z2);
    \draw[tn line, draw=exact] (0,-0.8) -- (2.3,-0.8);
  \end{scope}

  % The named colours: the physics ones, the MPS ones, then the page tokens.
  % `color=` and not a bare `\fill[\c]`, because several of these names are also
  % styles (coef, mpo, gate, centre), and TikZ resolves an option as a key before
  % it tries it as a colour -- a bare one silently fills with the style instead.
  \foreach \c [count=\i] in {ele,nuc,coef,exchange,exact,%
                              canl,canr,centre,mpo,gate,%
                              ttfg,ttmuted,ttaccent,ttgood,ttwarn,ttbad}
    \fill[color=\c] (\i*0.6-0.6,-2.1) rectangle ++(0.45,0.3);
\end{tikzpicture}
\end{document}
