% A specimen of types that name their indices, not a diagram.
%
% \tntype names a type, and tn index names its indices, each on a side, in an
% order: a chain's bulk has l, r and s, its ends one bond each. A picture
% that ends with a named index not drawn is an error. Several indices on one
% side leave the tensor apart, and two tensors beside each other are joined
% index by index in order: W has two left, joined to V's two right. A port is
% named by the index's name (P-ket-1:r) or its place in order (W-a-2:3), and
% \tnopen{rest} draws what is still free -- of a block, of each tensor whose
% type names its indices; of a tensor, all of its own; of a port, that one.
% A tree's site and a grid's take names the same way. An index may say which
% way it runs, name:side:in or :out, and a bond then runs from the end that is
% out to the end that is in, which is the way its arrow points: F's chain
% runs left to right, its physical indices out, and the bond \tnjoin closes
% runs back from F-5 to F-1 because that is the way its ends say.
\documentclass[border=5pt]{standalone}
\usepackage{tikz-tensors}
\tntype{bulk}{tn circle, tn index={l:left, r:right, s:down}}
\tntype{first}{tn circle, tn index={r:right, s:down}}
\tntype{last}{tn circle, tn index={l:left, s:down}}
\tntype{pair}{tn box, tn index={a:right, b:right, p:down}}
\tntype{split}{tn box, tn index={a:left, b:left, u:up, d:down, e:down}}
\tntype{iso}{tn triangle up, tn index={top:up, x:down, y:down}}
\tntype{flow}{tn circle, tn index={l:left:in, r:right:out, s:down:out}}
\tntype{cell}{tn circle, tn index={w:nw, n:ne, e:se, s:sw, p:down}}
\begin{document}
\begin{tikzpicture}
  \tnstack{P}{5}{ket}
  \tnlayer{P}{ket}{first/$A$, 3*bulk/$A$, last/$A$}
  \tnconnect[apart={P-ket-1:r}]{P}
  \tnjoin{P-ket-1:r}{P-ket-2:l}
  \tnopen[label=$n_{#1}$]{rest}{P}
  \tnshow{P-ket-1, P-ket-3}
  \tnbreak
  \tnstack{W}{2}{a}
  \tnlayer{W}{a}{pair/$V$, split/$W$}
  \tnconnect{W}
  \tnopen{down}{W-a-2}
  \tnopen{rest}{W-a-2:3, W-a-1}
  \tnshow{W-a-2}
  \tnstack[tree]{T}{2}{top, leaves}
  \tnlayer{T}{top}{iso/$V$/2}
  \tnlayer{T}{leaves}{2*tn circle/$x$}
  \tnconnect[apart={leaves}]{T}
  \tnopen{rest}{T}
  \tnopen{down}{T}
  \tnbreak
  \tnstack{F}{5}{ket}
  \tnlayer{F}{ket}{5*flow/$F$}
  \tnconnect[along=tn arrow]{F}
  \tnjoin[edge=tn arrow]{F-ket-1:l}{F-ket-5:r}
  \tnopen[edge=tn arrow]{rest}{F}
  \tnbreak
  \tngrid{G}{2}{2}{cell/$T$}
  \tnconnect{G}
  \tnopen{rest}{G-1-1, G-2-1, G-1-2, G-2-2}
  \tnshow{G-1-1}
\end{tikzpicture}
\end{document}
