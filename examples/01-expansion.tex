%% Expansion in a basis
%
% The two-electron expansion and its Hartree-Fock restriction: C on the basis
% functions phi, which carry the positions; HF cuts C into one c per electron.
%
% Two stacks of two sites, the array above the basis functions. C is one entry
% across both sites, so it is as wide as the two c it is cut into; the
% positions are the open indices of the lower layer, wavy because they are
% continuous, and i and j are put on the slots they run between. The basis
% functions belong to different electrons, and HF's two c to different
% orbitals: those layers are `apart', with no index along them.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\input{notation}
\begin{document}
\begin{tikzpicture}
  \tnstack{X}{2}{array, basis}
  \tnlayer{X}{array}{coefwide/$C$/2}
  \tnlayer{X}{basis}{2*fn/$\varphi$}
  \tnconnect[apart=basis]{X}
  \tnopen[edge=cont, label=$\mathbf r_{#1}$]{down}{X}
  \tnmid[left]{X-array-1-slot}{X-basis-1-slot}{$i$}
  \tnmid[right]{X-array-2-slot}{X-basis-2-slot}{$j$}
  \tnapprox
  \tnstack{Y}{2}{array, basis}
  \tnlayer{Y}{array}{2*coef/$c$}
  \tnlayer{Y}{basis}{2*fn/$\varphi$}
  \tnconnect[apart={array, basis}]{Y}
  \tnopen[edge=cont, label=$\mathbf r_{#1}$]{down}{Y}
  \tnmid[left]{Y-array-1-slot}{Y-basis-1-slot}{$i$}
  \tnmid[right]{Y-array-2-slot}{Y-basis-2-slot}{$j$}
\end{tikzpicture}
\end{document}
