\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}
  \node[coefwide, minimum width=22mm] (C) at (0.75,1.3) {$C$};
  \node[fn] (p1) at (0,0) {$\varphi$};
  \node[fn] (p2) at (1.5,0) {$\varphi$};
  \tnbond[disc]{(C.south -| p1) -- node[leg, left] {$i$} (p1)}
  \tnbond[disc]{(C.south -| p2) -- node[leg, right] {$j$} (p2)}
  \tnbond[cont]{(p1.south) -- ++(0,-0.9) node[leg, below] {$\mathbf r_1$}}
  \tnbond[cont]{(p2.south) -- ++(0,-0.9) node[leg, below] {$\mathbf r_2$}}
  \node at (2.75,0.55) {$\approx$};
  \node[coef] (c1) at (4.0,1.3) {$c$};
  \node[coef] (c2) at (5.5,1.3) {$c$};
  \node[fn] (q1) at (4.0,0) {$\varphi$};
  \node[fn] (q2) at (5.5,0) {$\varphi$};
  \tnbond[disc]{(c1) -- node[leg, left] {$i$} (q1)}
  \tnbond[disc]{(c2) -- node[leg, right] {$j$} (q2)}
  \tnbond[cont]{(q1.south) -- ++(0,-0.9) node[leg, below] {$\mathbf r_1$}}
  \tnbond[cont]{(q2.south) -- ++(0,-0.9) node[leg, below] {$\mathbf r_2$}}
