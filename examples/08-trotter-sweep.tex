% A Trotter step as it is actually applied: one gate at a time, along the chain.
%
% The two-layer picture -- all the odd bonds, then all the even ones -- is how
% the splitting is derived. It is not how it is run. A gate acts on two sites,
% so it goes on by itself, and the sweep runs bond by bond.
%
% First order is one pass: (1,2), (2,3), (3,4), (4,5), each with the full step.
% Second order is the same pass at half a step and then the same bonds in
% reverse. That sequence is its own reverse, which is what kills the leading
% error term, and the picture shows the symmetry rather than asserting it. One
% more pass is the whole price.
%
% The state is drawn in a definite mixed canonical form, which is what it is
% when the step begins. It does not stay in that one: the orthogonality centre
% travels with the sweep, so after a gate on (i, i+1) and the truncation that
% follows it sits ready for (i+1, i+2). That is the practical reason the order
% is a sweep and not two layers -- in two layers the centre would have to be
% dragged back across the chain between them.
%
% Read downward: the state at the top, one layer per gate under it. No gate has
% legs of its own; the sites it does not touch are `|' in its layer, so every
% physical index runs the whole way down and a gate is laid on the two it acts
% on. The order of the step is written beside the middle of its sweep.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\input{notation}
\begin{document}
\begin{tikzpicture}
  \tnstack{S}{5}{ket, g1, g2, g3, g4}
  \tnlayer{S}{ket}{2*canl/$A$, centre/$C$, 2*canr/$B$}
  \tnlayer{S}{g1}{gate/$U(\delta t)$/2, 3*|}
  \tnlayer{S}{g2}{|, gate/$U(\delta t)$/2, 2*|}
  \tnlayer{S}{g3}{2*|, gate/$U(\delta t)$/2, |}
  \tnlayer{S}{g4}{3*|, gate/$U(\delta t)$/2}
  \tnconnect[along=gauge]{S}
  \tnopen[gauge]{left}{S-ket-1}
  \tnopen[gauge]{right}{S-ket-5}
  \tnopen{down}{S}
  \tnmid[right]{S-g2-6-slot}{S-g3-6-slot}{$O(\delta t^2)$ a step}
  \tnbreak
  \tnstack{T}{5}{ket, g1, g2, g3, g4, g5, g6, g7, g8}
  \tnlayer{T}{ket}{2*canl/$A$, centre/$C$, 2*canr/$B$}
  \tnlayer{T}{g1}{gate/{$U(\delta t/2)$}/2, 3*|}
  \tnlayer{T}{g2}{|, gate/{$U(\delta t/2)$}/2, 2*|}
  \tnlayer{T}{g3}{2*|, gate/{$U(\delta t/2)$}/2, |}
  \tnlayer{T}{g4}{3*|, gate/{$U(\delta t/2)$}/2}
  \tnlayer{T}{g5}{3*|, gate/{$U(\delta t/2)$}/2}
  \tnlayer{T}{g6}{2*|, gate/{$U(\delta t/2)$}/2, |}
  \tnlayer{T}{g7}{|, gate/{$U(\delta t/2)$}/2, 2*|}
  \tnlayer{T}{g8}{gate/{$U(\delta t/2)$}/2, 3*|}
  \tnconnect[along=gauge]{T}
  \tnopen[gauge]{left}{T-ket-1}
  \tnopen[gauge]{right}{T-ket-5}
  \tnopen{down}{T}
  \tnmid[right]{T-g4-6-slot}{T-g5-6-slot}{$O(\delta t^3)$ a step}
\end{tikzpicture}
\end{document}
