% TEBD as it is actually applied: one gate at a time, up the chain and back
% down.
%
% The two-layer picture -- all the odd bonds, then all the even ones -- is the
% same operator grouped by what commutes, and it is how the splitting is derived.
% It is not how it is run. A gate acts on two sites, so it is applied on its own,
% and the sweep goes bond by bond: (1,2), (2,3), (3,4), (4,5), and then the same
% bonds in reverse. That sequence is its own reverse, which is what makes it
% second order, and the picture shows the symmetry rather than asserting it.
%
% Running it this way is what keeps the orthogonality centre where the next gate
% needs it. The centre travels with the sweep -- after a gate on (i, i+1) and the
% truncation that follows, it sits ready for the gate on (i+1, i+2) -- so no one
% canonical form labels the whole picture, and the sites are drawn plain. That
% is the practical reason the order is a sweep and not two layers: in two layers
% the centre would have to be dragged back across the chain between them.
%
% Read downward: the state at the top, the first gate applied just under it, the
% last at the bottom. No gate has legs of its own; the physical indices run the
% whole way down and a gate is laid on the two it acts on.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\begin{document}
% the sites take the circle convention, `tn round': this picture has no
% functions in it to tell a circle from, and a plain tensor is what they are
\tikzset{site/.style={coef, tn round}}
\begin{tikzpicture}[x=1.45cm]
  \tnchain{S}{(0,0)}{(4,0)}{site/{}/c, site/{}/c, site/{}/c, site/{}/c, site/{}/c}
  \foreach \i [evaluate=\i as \j using int(\i+1)] in {1,...,4}{
    \tnbond[disc]{(S\i) -- (S\j)}%
  }
  \node[leg, anchor=east] at (-0.35,0) {$|\psi\rangle$};
  \tnlegs{92mm}{S1-leg,S2-leg,S3-leg,S4-leg,S5-leg}

  % up the chain, then the same bonds back down
  \foreach \b/\d [count=\k] in {1/9, 2/19, 3/29, 4/39, 4/49, 3/59, 2/69, 1/79}{
    \pgfmathtruncatemacro{\c}{\b+1}
    \tngate{G\k}{$U(\delta t/2)$}{S\b-leg,S\c-leg}{\d mm}%
  }
\end{tikzpicture}
\end{document}
