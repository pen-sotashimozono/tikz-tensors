% A tree tensor network on eight sites: four tiers, from the physical indices
% at the bottom to the root at the top.
%
% Each isometry W takes two indices up to one: the triangle points toward the
% root, which is the direction it is isometric in, and it is as wide as the
% sites under it. Its one upper index leaves from the apex, in the middle of
% those sites, and meets the tensor above there -- the grid joins two tensors
% of different widths once, in the middle of the sites they share, so no
% position in this file is worked out.
%
% The root is the centre, spanning the whole tree, the one tensor that is not
% isometric. Move the centre down a branch and the triangles it passes through
% turn over; the notation has the shape for that (a triangle pointing down)
% but this picture keeps it at the top.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\input{notation}
\begin{document}
\begin{tikzpicture}
  \tnstack{T}{8}{root, w4, w2}
  \tnlayer{T}{root}{centre/$C$/8}
  \tnlayer{T}{w4}{iso/$W$/4, iso/$W$/4}
  \tnlayer{T}{w2}{iso/$W$/2, iso/$W$/2, iso/$W$/2, iso/$W$/2}
  \tnconnect[apart={root, w4, w2}]{T}
  \tnopen{down}{T}
  \foreach \i in {1,...,8} \tnput[below]{T-\i-down}{$\sigma_{\i}$};
\end{tikzpicture}
\end{document}
