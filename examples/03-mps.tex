% A matrix product state: arrays A on the chain, bonds (disc) between them,
% occupation indices n_i (disc) down -- a finite basis, so no wavy legs.
%
% The same chain twice, in the two conventions a tensor is drawn in. Square is
% this package's default, because elsewhere it tells an array of numbers from a
% function of position, which is a circle. Here there are no functions, so there
% is nothing to tell apart, and a great deal of the literature draws every
% tensor of a network as a circle -- that is the lower row, which is `tn round`
% and nothing else: same style, same colour, same meaning.
%
% Shape does carry meaning once a chain is in canonical form, where a triangle
% says isometric and a diamond says centre (04, 05). That is a different claim
% from this one, and it is not what `tn round` touches.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}
  \foreach \i/\l in {1/1,2/2,3/3,5/N} {
    \node[coef] (A\i) at (1.4*\i,0) {$A$};
    \draw[disc] (A\i) -- ++(0,-0.9) node[leg, below] {$n_{\l}$};
  }
  \draw[disc] (A1) -- (A2); \draw[disc] (A2) -- (A3);
  \draw[disc] (A3) -- ++(0.55,0); \draw[disc] (A5) -- ++(-0.55,0);
  \node at (5.6,0) {$\cdots$};
  \node[leg, above] at (2.1,0) {$D$};

  \begin{scope}[yshift=-2.4cm]
    \foreach \i/\l in {1/1,2/2,3/3,5/N} {
      \node[coef, tn round] (B\i) at (1.4*\i,0) {$A$};
      \draw[disc] (B\i) -- ++(0,-0.9) node[leg, below] {$n_{\l}$};
    }
    \draw[disc] (B1) -- (B2); \draw[disc] (B2) -- (B3);
    \draw[disc] (B3) -- ++(0.55,0); \draw[disc] (B5) -- ++(-0.55,0);
    \node at (5.6,0) {$\cdots$};
    \node[leg, above] at (2.1,0) {$D$};
  \end{scope}
\end{tikzpicture}
\end{document}
