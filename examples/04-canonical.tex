% A matrix product state in mixed canonical form, with a two-site gate about to
% be applied: one step of TEBD. The triangles say which tensors are isometric --
% canl to the left of the centre, canr to the right -- and the arrows on the
% bonds say which way the gauge runs. The centre is the one tensor that is not
% isometric, and it is also the one drawn as a diamond: shape and colour both
% single it out, which is the point, because the norm and every local
% expectation value can be read off it alone. The caps are the fixed points
% that close an infinite chain, one colour each end so a picture of a cut chain
% says which cut it is.
%
% The gate sits on the physical legs, which is the side they point to; it is the
% physical indices it acts on, not the bonds.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}[x=1.35cm]
  % the chain: isometric to both sides, orthogonality centre in the middle
  \node[env]    (EL) at (-0.9,0) {};
  \node[canl]   (A1) at (0,0)    {$A$};
  \node[canl]   (A2) at (1,0)    {$A$};
  \node[centre] (C)  at (2,0)    {$C$};
  \node[canr]   (B1) at (3,0)    {$B$};
  \node[canr]   (B2) at (4,0)    {$B$};
  \node[envc]   (ER) at (4.9,0)  {};

  % bonds, carrying the gauge direction: everything points at the centre
  \draw[bond] (EL) -- (A1);  \draw[bond] (A1) -- (A2);  \draw[bond] (A2) -- (C);
  \draw[bond] (ER) -- (B2);  \draw[bond] (B2) -- (B1);  \draw[bond] (B1) -- (C);

  % the gate, on the physical legs of the two sites it acts on
  \node[gate, minimum width=23mm] (U) at (2.5,-1.1) {$e^{-i\,\delta t\,\hat h}$};
  \tngatelegs{U}{north}{C,B1}

  % physical legs: through the gate where there is one, straight down elsewhere
  \foreach \n/\l in {A1/1,A2/2,B2/5}
    \draw[disc] (\n) -- ++(0,-2.1) node[leg, below] {$n_{\l}$};
  \foreach \n/\l in {C/3,B1/4}
    \draw[disc] (U.south -| \n) -- ++(0,-0.55) node[leg, below] {$n_{\l}$};

  \node[leg, above=1pt] at (0.5,0.25) {$D$};
  \node[leg] at (-0.9,-0.75) {$\mathcal{E}_{\mathrm{L}}$};
  \node[leg] at (4.9,-0.75)  {$\mathcal{E}_{\mathrm{R}}$};
\end{tikzpicture}
\end{document}
