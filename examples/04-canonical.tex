% A finite matrix product state in mixed canonical form, with a two-site gate
% about to be applied: one step of TEBD. The triangles say which tensors are
% isometric -- canl to the left of the centre, canr to the right -- and the
% arrows on the bonds say which way the gauge runs. The centre is the one
% tensor that is not isometric, and it is also the one drawn as a diamond:
% shape and colour both single it out, which is the point, because the norm and
% every local expectation value can be read off it alone.
%
% A stack of two layers, the state and the gate. The arrows are not written:
% a bond takes its direction from the triangles it joins, away from one
% pointing right and into one pointing left, so every arrow runs toward the
% centre wherever the centre is put. The physical index leaves each triangle
% at the corner of its flat side and continues that side's line.
%
% The gate has no legs of its own. The indices run past it and a node covers an
% edge, so it interrupts them -- which is what applying a gate to those indices
% looks like. The sites it does not touch are `|' in its layer, so all five
% indices run on to one depth.
%
% Finite, so the chain simply stops: the outer indices are open, with nothing at
% the far end, and point inward with the gauge.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\input{notation}
\begin{document}
\begin{tikzpicture}
  \tnstack{S}{5}{ket, gate}
  \tnlayer{S}{ket}{canl/$A$, canl/$A$, centre/$C$, canr/$B$, canr/$B$}
  \tnlayer{S}{gate}{|, |, gate/$e^{-i\,\delta t\,\hat h}$/2, |}
  \tnconnect[along=gauge]{S}
  \tnopen[gauge]{left}{S-ket-1}
  \tnopen[gauge]{right}{S-ket-5}
  \tnopen{down}{S}
  \tnmid{S-ket-1}{S-ket-2}{$D$}
  \foreach \i in {1,...,5} \tnput[below]{S-\i-down}{$n_{\i}$};
\end{tikzpicture}
\end{document}
