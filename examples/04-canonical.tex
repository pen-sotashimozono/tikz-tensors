% A finite matrix product state in mixed canonical form, with a two-site gate
% about to be applied: one step of TEBD. The triangles say which tensors are
% isometric -- canl to the left of the centre, canr to the right -- and the
% arrows on the bonds say which way the gauge runs. The centre is the one
% tensor that is not isometric, and it is also the one drawn as a diamond:
% shape and colour both single it out, which is the point, because the norm and
% every local expectation value can be read off it alone.
%
% \tnchain places the row and settles, once, where each site's physical index
% leaves it -- the bottom corner of the flat side, mirrored between the two
% gauges, and the diamond's own south. After that nothing here names a corner
% or a site's x again: the legs, the gate and the gate's legs all connect to the
% same S<i>-leg coordinates.
%
% Finite, so the chain simply stops: the outer indices are open, drawn as stubs
% with nothing at the far end, and drawn inward so their arrowheads run toward
% the centre like every other bond.
%
% The page reads downward in the order the parts are applied: the state, then
% the gate on the physical indices of the two sites it acts on -- the physical
% indices, not the bonds -- then the legs that come out of it, which end level
% with the three that passed the gate by.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}[x=1.35cm]
  \tnchain{S}{(0,0)}{canl/$A$/l, canl/$A$/l, centre/$C$/c,
                     canr/$B$/r, canr/$B$/r}

  % bonds, carrying the gauge direction: everything points at the centre
  \tnbond[gauge]{(S1) -- (S2)}  \tnbond[gauge]{(S2) -- (S3)}
  \tnbond[gauge]{(S5) -- (S4)}  \tnbond[gauge]{(S4) -- (S3)}
  \tnbond[gauge]{([xshift=-5mm]S1.west) -- (S1)}
  \tnbond[gauge]{([xshift=5mm]S5.east) -- (S5)}
  \node[leg, above=1pt] at (0.5,0.25) {$D$};

  % the gate, sized and placed from the two indices it acts on
  \tngate{U}{$e^{-i\,\delta t\,\hat h}$}{S3-leg,S4-leg}{-1.15}
  \tngatelegs{U}{north}{S3-leg,S4-leg}

  % physical legs, all ending level: straight down where the gate is not,
  % out of the gate where it is
  \tnlegs{-2.3}{S1-leg,S2-leg,S5-leg}
  \tnbond[disc]{(U.south -| S3-leg) -- (S3-leg |- 0,-2.3)}
  \tnbond[disc]{(U.south -| S4-leg) -- (S4-leg |- 0,-2.3)}
  \foreach \i in {1,...,5}
    \node[leg, below] at (S\i-leg |- 0,-2.3) {$n_{\i}$};
\end{tikzpicture}
\end{document}
