% The superblock of two-site DMRG: the effective Hamiltonian the eigensolver is
% handed at one step of a sweep.
%
% It is the fixed point's left-hand side with the state removed. Take L from
% 07 -- the tall tensor carrying one index into each layer and one into the
% Hamiltonian -- take the A and its conjugate out of the column, and what is
% left is L with its bra and ket indices free. Put that on the left, its mirror
% on the right, and the two MPO tensors of the sites being optimised between
% them, and the picture is H_eff.
%
% So the free indices are where the state used to be, and they point at the gap
% it would fill: the two stubs above and the two legs of W between them are the
% four indices of the two-site tensor that comes out, the two below and the two
% legs under W are the four that go in. The gaps are left empty on purpose --
% drawing Theta into one would make this a picture of one application rather
% than of the operator.
%
% Every index is the same length and the four of each half end at one height,
% because nothing here is said by how long a line is. Nothing in the picture is
% the state: that is the point of the form, and why the solver never sees the
% sites that were contracted away.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}
  \node[tn node, minimum width=7mm, minimum height=22mm] (L) at (0,0)   {$L$};
  \node[tn node, minimum width=7mm, minimum height=22mm] (R) at (4.3,0) {$R$};
  \node[mpo] (W1) at (1.7,0) {$W$};
  \node[mpo] (W2) at (2.6,0) {$W$};

  % the Hamiltonian's own bond, straight through
  \tnbond[disc]{(L) -- (W1)}  \tnbond[disc]{(W1) -- (W2)}  \tnbond[disc]{(W2) -- (R)}

  % the indices the state used to be attached to, free and pointing at the gap
  \foreach \s/\p in {1/{'}, -1/{}} {
    \tnbond[disc]{([yshift=\s*8mm]L.east) -- ++(0.45,0)}
    \tnbond[disc]{([yshift=\s*8mm]R.west) -- ++(-0.45,0)}
    \tnbond[disc]{(W1) -- (W1 |- 0,\s*0.8)}
    \tnbond[disc]{(W2) -- (W2 |- 0,\s*0.8)}
  }
  \tnput[above]{[yshift=8mm,xshift=4.5mm]L.east}{$a'$}
  \tnput[above]{W1 |- 0,0.8}{$\sigma_1'$}
  \tnput[above]{W2 |- 0,0.8}{$\sigma_2'$}
  \tnput[above]{[yshift=8mm,xshift=-4.5mm]R.west}{$b'$}
  \tnput[below]{[yshift=-8mm,xshift=4.5mm]L.east}{$a$}
  \tnput[below]{W1 |- 0,-0.8}{$\sigma_1$}
  \tnput[below]{W2 |- 0,-0.8}{$\sigma_2$}
  \tnput[below]{[yshift=-8mm,xshift=-4.5mm]R.west}{$b$}
\end{tikzpicture}
\end{document}
