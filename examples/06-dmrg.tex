% The superblock of two-site DMRG: the effective Hamiltonian the eigensolver is
% handed at one step of a sweep.
%
% Everything to the left of the two active sites has been contracted into L and
% everything to the right into R, so each carries three indices and no more --
% the bra's bond, the Hamiltonian's bond, the ket's bond. Between them sit the
% two MPO tensors of the sites being optimised, each with a physical index in
% from below and out above.
%
% Read it as a matrix: the four indices below and to the lower sides are what it
% acts on, the four above and to the upper sides are what it returns, and the
% eigenvector of that matrix is the two-site wavefunction. Nothing in the
% picture is the state itself -- that is the point of the form, and why the
% solver never sees the sites that were contracted away.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}[x=1.3cm]
  \node[tn node, minimum width=7mm, minimum height=24mm] (L) at (0,0)   {$L$};
  \node[tn node, minimum width=7mm, minimum height=24mm] (R) at (3.4,0) {$R$};
  \node[mpo] (W1) at (1.3,0) {$W$};
  \node[mpo] (W2) at (2.1,0) {$W$};

  % the Hamiltonian's own bond, straight through the middle
  \tnbond[disc]{(L) -- (W1)}  \tnbond[disc]{(W1) -- (W2)}  \tnbond[disc]{(W2) -- (R)}

  % the bra's bonds above and the ket's below, free at both ends because the
  % state itself is not in this picture
  \coordinate (ap) at ([yshift=9mm,xshift=7mm]L.east);
  \coordinate (am) at ([yshift=-9mm,xshift=7mm]L.east);
  \coordinate (bp) at ([yshift=9mm,xshift=-7mm]R.west);
  \coordinate (bm) at ([yshift=-9mm,xshift=-7mm]R.west);
  \tnbond[disc]{([yshift=9mm]L.east) -- (ap)}
  \tnbond[disc]{([yshift=-9mm]L.east) -- (am)}
  \tnbond[disc]{([yshift=9mm]R.west) -- (bp)}
  \tnbond[disc]{([yshift=-9mm]R.west) -- (bm)}
  \tnput[right]{ap}{$a'$}  \tnput[right]{am}{$a$}
  \tnput[left]{bp}{$b'$}   \tnput[left]{bm}{$b$}

  % the physical indices of the two sites: in from below, out above
  \foreach \n/\i in {W1/1, W2/2} {
    \coordinate (\n u) at ([yshift=11mm]\n.north);
    \coordinate (\n d) at ([yshift=-11mm]\n.south);
    \tnbond[disc]{(\n.north) -- (\n u)}
    \tnbond[disc]{(\n.south) -- (\n d)}
  }
  \tnput[above]{W1u}{$\sigma_1'$}  \tnput[above]{W2u}{$\sigma_2'$}
  \tnput[below]{W1d}{$\sigma_1$}   \tnput[below]{W2d}{$\sigma_2$}
\end{tikzpicture}
\end{document}
