% The superblock of two-site DMRG: the effective Hamiltonian the eigensolver is
% handed at one step of a sweep.
%
% Everything to the left of the two active sites has been contracted into L and
% everything to the right into R, so each carries three indices and no more --
% the bra's bond, the Hamiltonian's bond, the ket's bond. Between them sit the
% two MPO tensors of the sites being optimised.
%
% The four indices below end at one depth and are spaced to take a tensor, and
% the four above do the same, because that is what this object is for: the
% two-site wavefunction Theta, with legs (a, sigma_1, sigma_2, b), drops into
% the gap underneath and the picture closes into H_eff Theta. The space is left
% empty on purpose -- drawing Theta in would make this a picture of one
% application rather than of the operator.
%
% Nothing here is the state. That is the point of the form, and why the solver
% never sees the sites that were contracted away: it works in the four indices
% left over, not in the length of the chain.
\documentclass[border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{tikz-tensors}
\begin{document}
\begin{tikzpicture}
  \node[tn node, minimum width=7mm, minimum height=22mm] (L) at (0,0)   {$L$};
  \node[tn node, minimum width=7mm, minimum height=22mm] (R) at (3.3,0) {$R$};
  \node[mpo] (W1) at (1.15,0) {$W$};
  \node[mpo] (W2) at (2.15,0) {$W$};

  % the Hamiltonian's own bond: short, because its length says nothing
  \tnbond[disc]{(L) -- (W1)}  \tnbond[disc]{(W1) -- (W2)}  \tnbond[disc]{(W2) -- (R)}

  % the four indices below, at one depth and evenly spaced: Theta's legs
  \tnbond[disc]{([yshift=-7mm]L.east) -| (0.5,-1.55)}
  \tnbond[disc]{(W1.south) -- (1.15,-1.55)}
  \tnbond[disc]{(W2.south) -- (2.15,-1.55)}
  \tnbond[disc]{([yshift=-7mm]R.west) -| (2.8,-1.55)}
  \tnput[below]{0.5,-1.55}{$a$}        \tnput[below]{1.15,-1.55}{$\sigma_1$}
  \tnput[below]{2.15,-1.55}{$\sigma_2$} \tnput[below]{2.8,-1.55}{$b$}

  % and the four above, which is what it returns
  \tnbond[disc]{([yshift=7mm]L.east) -| (0.5,1.55)}
  \tnbond[disc]{(W1.north) -- (1.15,1.55)}
  \tnbond[disc]{(W2.north) -- (2.15,1.55)}
  \tnbond[disc]{([yshift=7mm]R.west) -| (2.8,1.55)}
  \tnput[above]{0.5,1.55}{$a'$}         \tnput[above]{1.15,1.55}{$\sigma_1'$}
  \tnput[above]{2.15,1.55}{$\sigma_2'$} \tnput[above]{2.8,1.55}{$b'$}
\end{tikzpicture}
\end{document}
