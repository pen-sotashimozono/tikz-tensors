% tikz-tensors-layout.tex -- the abstract layout. What every layout provides;
% the record of each tensor a layout places, its ports, and which of them are
% taken; and the commands a figure uses on any layout -- \tnconnect, \tnopen,
% \tnjoin, \tnopenswap -- each of which hands its work to the layout that
% placed the tensors.
%
% A layout is a kind of block of the canvas: a stack, a tree, a grid. A kind is
% declared with \tn@kind{<kind>}{<parent>} and implements operations as macros
% \tn@<kind>@<op>. An operation a kind does not define is its parent's (a tree
% is a stack that sizes and routes its own way), and one that no kind in the
% chain defines is an error naming both. The operations:
%
%   geometry{<block>}              its pitch and rise (a layout of slots)
%   connect{<block>}               every index the layout implies, styled by
%                                  \tn@along and \tn@down
%   openblock{<style>}{<block>}    every free index of the block toward \tn@dir
%   opennode{<style>}{<tensor>}    the tensor's own indices toward \tn@dir
%   join{<style>}{<port>}{<port>}  one index between two ports
%   openswap{<style>}{<tensor>}    its two indices down, crossed
%   vport{<tensor>}{up|down}       the port up or down in column \tn@cx
%   ports{<tensor>}                \tn@portvisit{<tensor>}{<side>}{<key>} for
%                                  each of its ports on the sides it has
%   openport{<style>}{<tensor>}{<side>}{<key>}
%                                  that one port, opened outward
%   portkey{<tensor>}{<side>}{<n>} the key of its <n>th port on <side>
%   mapindex{<tensor>}             the port of each index its type names
%
% A new layout is a file that declares its kind, a command that places a block
% (\tn@block, tikz-tensors-canvas.tex), places its tensors (\tn@place,
% tikz-tensors-nodes.tex) and records them (\tn@record below), and the
% operations it supports. Nothing outside the file needs to change.

% ---- kinds and operations ----------------------------------------------------
\def\tn@kind#1#2{\tn@set{kind@#1@parent}{#2}}
% \tn@op{<block>}{<op>}<arguments>: the operation of the block's kind
\def\tn@op#1#2{%
  \ifcsname tn@#1@kind\endcsname\else
    \PackageError{tikz-tensors}{`#1' is not a stack, tree or grid}{}\fi
  \edef\tn@opkind{\tn@get{#1@kind}}%
  \tn@tokens{#1}%
  \expandafter\tn@resolve\expandafter{\tn@opkind}{#2}\tn@fn}
\def\tn@resolve#1#2{%
  \ifcsname tn@#1@#2\endcsname
    \expandafter\let\expandafter\tn@fn\csname tn@#1@#2\endcsname
  \else
    \edef\tn@rpk{\tn@get{kind@#1@parent}}%
    \ifx\tn@rpk\@empty
      \PackageError{tikz-tensors}{A \tn@opkind\space cannot do `#2'}{}%
      \let\tn@fn\relax
    \else \expandafter\tn@resolve\expandafter{\tn@rpk}{#2}\fi
  \fi}
% the operation of the block a tensor is in
\def\tn@nop#1#2{\edef\tn@nopb{\tn@get{n@#1@s}}\tn@op{\tn@nopb}{#2}}

% ---- the size of a block --------------------------------------------------------
% A block is drawn at the notation's distances and sizes times factors of its
% own, given when it is placed -- never a length, so a figure that is drawn
% smaller still says nothing a notation should:
%   pitch=<f>   between its sites (pitch, tree pitch, grid pitch)
%   rise=<f>    between its layers (rise, tree rise)
%   size=<f>    its tensors, and the slot an absent one leaves (slot,
%               env width)
%   stub=<f>    the least an open index of it shows
%   scale=<f>   all four at once; another key multiplies it further
% A placing command forwards these keys to /tn@factor, which is internal;
% \tn@scaleblock{<block>} keeps the block's tokens, its notation's times its
% factors; \tn@tokens{<block>} makes them the ones in use, which every
% operation on the block does first.
\pgfkeys{/tn@factor/.is family,
  /tn@factor/scale/.code={\def\tn@fsc{#1}}, /tn@factor/pitch/.code={\def\tn@fp{#1}},
  /tn@factor/rise/.code={\def\tn@fr{#1}}, /tn@factor/size/.code={\def\tn@fs{#1}},
  /tn@factor/stub/.code={\def\tn@ft{#1}}}
\def\tn@factorsreset{\def\tn@fsc{1}\def\tn@fp{1}\def\tn@fr{1}\def\tn@fs{1}\def\tn@ft{1}}
\tn@factorsreset
\def\tn@scaletoken#1#2#3{% block, token, factor
  \tn@len\tn@tv{\csname tn@base@#2\endcsname*(#3)*(\tn@fsc)}\tn@set{#1@t@#2}{\tn@tv}}
\def\tn@scaleblock#1{%
  \tn@scaletoken{#1}{pitch}{\tn@fp}\tn@scaletoken{#1}{treepitch}{\tn@fp}%
  \tn@scaletoken{#1}{gpitch}{\tn@fp}%
  \tn@scaletoken{#1}{rise}{\tn@fr}\tn@scaletoken{#1}{treerise}{\tn@fr}%
  \tn@scaletoken{#1}{slot}{\tn@fs}\tn@scaletoken{#1}{envw}{\tn@fs}%
  \tn@scaletoken{#1}{stub}{\tn@ft}%
  \pgfmathsetmacro\tn@tv{(\tn@fs)*(\tn@fsc)}\tn@set{#1@t@size}{\tn@tv}%
  \tn@factorsreset}
\def\tn@tokens#1{%
  \edef\tn@pitch{\tn@get{#1@t@pitch}}\edef\tn@treepitch{\tn@get{#1@t@treepitch}}%
  \edef\tn@gpitch{\tn@get{#1@t@gpitch}}\edef\tn@rise{\tn@get{#1@t@rise}}%
  \edef\tn@treerise{\tn@get{#1@t@treerise}}%
  \edef\tn@slot{\tn@get{#1@t@slot}}\edef\tn@envw{\tn@get{#1@t@envw}}%
  \edef\tn@stub{\tn@get{#1@t@stub}}}
% The options a tensor of block <#1> is drawn with: its type, at the block's
% size, as \tn@opt -- and at <#2> times that, a layer's own size= (1 if none).
\def\tn@sized#1#2{%
  \pgfmathsetmacro\tn@sz{\tn@get{#1@t@size}*(#2)}%
  \ifdim\tn@sz pt=1pt \edef\tn@opt{\tn@sty}\else \edef\tn@opt{\tn@sty, tn size=\tn@sz}\fi}

% ---- the record of a tensor --------------------------------------------------
% \tn@record{<tensor>}{<block>}{<a>}{<b>}{<k>}{<kb>}
% What a layout knows of a tensor once it is placed: its block, the sites a..b
% across and the layers (or rows) k..kb down that it covers. Its type's index
% properties are kept by \tn@place. Each block placed is a new generation, so
% that a name used again in a later picture is a new tensor with free ports.
\newcount\tn@gen
% The block keeps the list of its tensors (<block>@nodes). A tensor whose type
% names its indices (tn index) has each put on a port of its own by the
% layout, and goes on the list the picture checks when it ends.
\def\tn@record#1#2#3#4#5#6{%
  \tn@set{n@#1@s}{#2}\tn@set{n@#1@a}{#3}\tn@set{n@#1@b}{#4}%
  \tn@set{n@#1@k}{#5}\tn@set{n@#1@kb}{#6}\tn@set{n@#1@g}{\the\tn@gen}%
  \edef\tn@rl{\tn@get{#2@nodes}}%
  \ifx\tn@rl\@empty \tn@set{#2@nodes}{#1}\else \tn@set{#2@nodes}{\tn@rl,#1}\fi
  \ifcsname tn@n@#1@index\endcsname \edef\tn@rl{\tn@get{n@#1@index}}%
    \ifx\tn@rl\@empty\else
      \tn@op{#2}{mapindex}{#1}%
      \ifx\tn@ranked\@empty \xdef\tn@ranked{#1}\else \xdef\tn@ranked{\tn@ranked,#1}\fi
    \fi\fi}
\def\tn@newblock#1#2{% block, kind
  \global\advance\tn@gen by 1
  \tn@set{#1@kind}{#2}\tn@set{#1@g}{\the\tn@gen}\tn@set{#1@nodes}{}}
\def\tn@isnode#1#2{% is <#1> a tensor placed by a layout? else an error for <#2>
  \ifcsname tn@n@#1@s\endcsname\else
    \PackageError{tikz-tensors}{#2: `#1' is not a tensor on a stack, tree or
      grid}{}\fi}

% ---- ports -------------------------------------------------------------------
% A tensor's indices are ports: on a side (up, down, left, right on a stack;
% the four directions of the lattice and down on a grid), and where on that
% side -- the column or layer, in lattice units. A port is a point just inside
% the tensor, where an index starts, and the border the index crosses.
%
% Each port is drawn once. \tn@use marks one taken; \tn@ifused asks; \tn@free
% is an error if it is taken, for the commands that name a port themselves.
% A layout opening every free index of a block skips the taken ones, so an
% index joined by hand is not opened as well.
\def\tn@key#1{\pgfmathsetmacro\tn@kk{#1}}
\def\tn@use#1#2#3{\tn@key{#3}\tn@set{u@\tn@get{n@#1@g}@#1@#2@\tn@kk}{1}}
\def\tn@ifused#1#2#3{\tn@key{#3}%
  \ifcsname tn@u@\tn@get{n@#1@g}@#1@#2@\tn@kk\endcsname
    \def\tn@isused{1}\else\def\tn@isused{0}\fi}
\def\tn@free#1#2#3#4{%
  \tn@ifused{#1}{#2}{#3}%
  \ifnum\tn@isused=1
    \PackageError{tikz-tensors}{#4: the index of `#1' on its #2 side (at
      \tn@kk) is drawn already}{An index is drawn once, by \string\tnconnect,
      \string\tnopen or \string\tnjoin.}\fi}

% Has tensor <#1> an index on side <#2> at all? \tn@hasl is 1 if its type
% does not name its indices (every side the layout gives it) or names one on
% that side, else 0. \tn@needleg is an error for a command that names one.
\def\tn@hasleg#1#2{%
  \ifcsname tn@n@#1@legs\endcsname \edef\tn@hl{\tn@get{n@#1@legs}}\else \def\tn@hl{}\fi
  \ifx\tn@hl\@empty \def\tn@hasl{1}%
  \else
    \edef\tn@hlt{\noexpand\in@{,#2,}{,\tn@hl,}}\tn@hlt
    \ifin@ \def\tn@hasl{1}\else \def\tn@hasl{0}\fi
  \fi}
\def\tn@needleg#1#2#3{\tn@hasleg{#1}{#2}%
  \ifnum\tn@hasl=0
    \PackageError{tikz-tensors}{#3: `#1' has no index on its #2 side; its type
      says tn index={\tn@get{n@#1@index}}}{}\fi}

% ---- named indices ------------------------------------------------------------
% A tensor whose type says tn index={<name>:<side>, ...} has those indices:
% n@<t>@ix@<name> is its side and n@<t>@ik@<name> its port's key, which the
% layout gives it (mapindex); n@<t>@io@<n> is the name of the <n>th, and
% n@<t>@isp@<side> is 1 when the indices on that side are not one per site or
% layer but spread across the tensor (\tn@spread, in the layout).
\def\tn@isnamed#1{\ifcsname tn@n@#1@index\endcsname
  \expandafter\ifx\csname tn@n@#1@index\endcsname\@empty \def\tn@named{0}%
  \else \def\tn@named{1}\fi \else \def\tn@named{0}\fi}
% each index of tensor <#1> in order: <#2>{<name>}{<side>}
\def\tn@eachindex#1#2{%
  \edef\tn@eil{\tn@get{n@#1@index}}%
  \foreach \tn@ei in \tn@eil{\expandafter\tn@eione\tn@ei\tn@end{#2}}}
\def\tn@eione#1:#2\tn@end#3{#3{#1}{#2}}
% The indices of <#1> on side <#2>, in order, as \tn@onside (names) and
% \tn@nonside (how many).
\newcount\tn@onc
\def\tn@sideindices#1#2{\gdef\tn@onside{}\global\tn@onc=0
  \edef\tn@sidewant{#2}%
  \tn@eachindex{#1}\tn@sidepick
  \edef\tn@nonside{\the\tn@onc}}
\def\tn@sidepick#1#2{\edef\tn@sidegot{#2}%
  \ifx\tn@sidegot\tn@sidewant \global\advance\tn@onc by 1
    \ifx\tn@onside\@empty \xdef\tn@onside{#1}\else \xdef\tn@onside{\tn@onside,#1}\fi\fi}
% A layout's mapindex puts index <#2> of <#1> on <#3> at key <#4>.
\newcount\tn@ixo
\def\tn@ixput#1#2#3#4{%
  \tn@set{n@#1@ix@#2}{#3}\tn@key{#4}\tn@set{n@#1@ik@#2}{\tn@kk}}
\def\tn@ixorder#1{\global\tn@ixo=0 \def\tn@ixon{#1}\tn@eachindex{#1}\tn@ixordone}
\def\tn@ixordone#1#2{\global\advance\tn@ixo by 1
  \tn@set{n@\tn@ixon @io@\the\tn@ixo}{#1}}

% ---- which way an index runs -------------------------------------------------------
% \tn@fluxof{<tensor>}{<side>}{<key>}: the way its port there runs, in, out or
% nothing said, as \tn@fx: the index's own (tn index), else its side's
% (tn flux) -- said, \tn@fxs is 1 -- else the way its shape leans (a
% triangle's), \tn@fxs 0. An empty <tensor> -- an environment, the dots --
% says nothing.
\def\tn@fluxof#1#2#3{\def\tn@fx{}\def\tn@fxs{1}\edef\tn@fxt{#1}%
  \ifx\tn@fxt\@empty\else
    \tn@isnamed{#1}%
    \ifnum\tn@named=1
      \tn@key{#3}\let\tn@fxk\tn@kk \gdef\tn@fxn{}%
      \tn@sideindices{#1}{#2}%
      \foreach \tn@fi in \tn@onside{\edef\tn@fik{\tn@get{n@#1@ik@\tn@fi}}%
        \ifdim\tn@fik pt=\tn@fxk pt \xdef\tn@fxn{\tn@fi}\fi}%
      \ifx\tn@fxn\@empty\else \ifcsname tn@n@#1@ifx@\tn@fxn\endcsname
        \edef\tn@fx{\tn@get{n@#1@ifx@\tn@fxn}}\fi\fi
    \fi
    \ifx\tn@fx\@empty \ifcsname tn@n@#1@fx@#2\endcsname
      \edef\tn@fx{\tn@get{n@#1@fx@#2}}\fi\fi
    \ifx\tn@fx\@empty \ifcsname tn@n@#1@wfx@#2\endcsname
      \edef\tn@fx{\tn@get{n@#1@wfx@#2}}\def\tn@fxs{0}\fi\fi
  \fi}
% \tn@orient{<tensor>}{<side>}{<key>}{<tensor>}{<side>}{<key>}{<command>}: an
% index from the first port to the second runs that way (\tn@rev 0) or the
% other (\tn@rev 1): from the end that is out to the end that is in, and from
% the first to the second when neither says. Where the ends disagree, what a
% type says overrules the way a shape leans; two ends that both say, both in
% or both out, are an error -- what flows out of one tensor flows into the
% next.
\def\tn@orient#1#2#3#4#5#6#7{%
  \tn@fluxof{#1}{#2}{#3}\let\tn@fxa\tn@fx \let\tn@fxsa\tn@fxs
  \tn@fluxof{#4}{#5}{#6}%
  % what each end asks for: 0 from the first to the second, 1 back, or none
  \def\tn@ra{}\ifx\tn@fxa\tn@sout \def\tn@ra{0}\fi \ifx\tn@fxa\tn@sin \def\tn@ra{1}\fi
  \def\tn@rb{}\ifx\tn@fx\tn@sin \def\tn@rb{0}\fi \ifx\tn@fx\tn@sout \def\tn@rb{1}\fi
  \ifx\tn@ra\@empty \ifx\tn@rb\@empty \def\tn@rev{0}\else \let\tn@rev\tn@rb \fi
  \else\ifx\tn@rb\@empty \let\tn@rev\tn@ra
  \else\ifx\tn@ra\tn@rb \let\tn@rev\tn@ra
  \else
    \ifnum\tn@fxsa\tn@fxs=11
      \PackageError{tikz-tensors}{#7: the index between `#1' (#2) and `#4' (#5)
        runs \tn@fx\space at both ends}{An index runs out of one tensor and
        into the other; tn flux and tn index say which way.}\def\tn@rev{0}%
    \else\ifnum\tn@fxsa=1 \let\tn@rev\tn@ra
    \else\ifnum\tn@fxs=1 \let\tn@rev\tn@rb
    \else \def\tn@rev{0}\fi\fi\fi
  \fi\fi\fi}

% ---- naming a port ---------------------------------------------------------------
% \tn@port{<tensor>:<which>[:<n>]}{<command>}: the port a figure names, as
% \tn@pn (the tensor), \tn@ps (its side) and \tn@pkey (its key) -- and the
% index's name as \tn@pnm, if it has one. <which> is
%   a side             the <n>th index on that side (the first unless said)
%   a number           the <n>th index of the tensor, in its type's order
%   a name             the index of that name (tn index)
\def\tn@sidenames{left,right,up,down,nw,ne,se,sw}
\def\tn@port#1#2{%
  \tn@pspec#1:::\tn@end
  \tn@isnode{\tn@pn}{#2}\tn@isnamed{\tn@pn}\def\tn@pnm{}%
  \edef\tn@pst{\noexpand\in@{,\tn@ps,}{,\tn@sidenames,}}\tn@pst
  \ifin@
    \tn@needleg{\tn@pn}{\tn@ps}{#2}%
    \ifx\tn@pk\@empty \def\tn@pk{1}\fi
    \ifnum\tn@named=1
      \tn@sideindices{\tn@pn}{\tn@ps}%
      \ifnum\tn@pk>\tn@nonside
        \PackageError{tikz-tensors}{#2: `\tn@pn' has \tn@nonside\space
          \ifnum\tn@nonside=1 index\else indices\fi\space on its \tn@ps\space
          side, not \tn@pk}{}\def\tn@pk{1}\fi
      \foreach \tn@x [count=\tn@xc] in \tn@onside{%
        \ifnum\tn@xc=\tn@pk \xdef\tn@pnmg{\tn@x}\fi}%
      \let\tn@pnm\tn@pnmg \edef\tn@pkey{\tn@get{n@\tn@pn @ik@\tn@pnm}}%
    \else
      \tn@nop{\tn@pn}{portkey}{\tn@pn}{\tn@ps}{\tn@pk}%
    \fi
  \else
    \ifnum\tn@named=0
      \PackageError{tikz-tensors}{#2: `\tn@ps' is not a side, and `\tn@pn'
        has no named indices (tn index)}{}%
    \else
      \edef\tn@pdig{\detokenize\expandafter{\romannumeral-0\tn@ps}}%
      \ifx\tn@pdig\@empty
        \ifcsname tn@n@\tn@pn @io@\tn@ps\endcsname
          \edef\tn@pnm{\tn@get{n@\tn@pn @io@\tn@ps}}%
        \else
          \PackageError{tikz-tensors}{#2: `\tn@pn' has no index number
            \tn@ps; it has \tn@get{n@\tn@pn @rank}}{}\edef\tn@pnm{\tn@get{n@\tn@pn @io@1}}%
        \fi
      \else \let\tn@pnm\tn@ps \fi
      \ifcsname tn@n@\tn@pn @ix@\tn@pnm\endcsname\else
        \PackageError{tikz-tensors}{#2: `\tn@pn' has no index `\tn@pnm'; its
          type says tn index={\tn@get{n@\tn@pn @index}}}{}%
        \edef\tn@pnm{\tn@get{n@\tn@pn @io@1}}\fi
      \edef\tn@ps{\tn@get{n@\tn@pn @ix@\tn@pnm}}%
      \edef\tn@pkey{\tn@get{n@\tn@pn @ik@\tn@pnm}}%
    \fi
  \fi}

% A bond the figure leaves out (\tnconnect's apart= naming a port): marked by
% port, as a port is marked drawn, and read where the layout would join it.
\def\tn@cut#1#2#3{\tn@key{#3}\tn@set{x@\tn@get{n@#1@g}@#1@#2@\tn@kk}{1}}
\def\tn@ifcut#1#2#3{\tn@key{#3}%
  \ifcsname tn@x@\tn@get{n@#1@g}@#1@#2@\tn@kk\endcsname
    \def\tn@iscut{1}\else\def\tn@iscut{0}\fi}

% ---- a tensor's indices, counted ----------------------------------------------
% \tn@ports{<tensor>}{<visit>}: <visit>{<tensor>}{<side>}{<key>} on each port
% of the tensor -- its named indices in order, with \tn@pnm the name, or the
% ports the layout gives it on every side, with \tn@pnm empty.
% \tn@drawn{<tensor>} counts the ones drawn, as \tn@ndrawn; \tn@nports all.
\newcount\tn@ndr \newcount\tn@npt
\def\tn@ports#1#2{\let\tn@portvisit#2\tn@isnamed{#1}%
  \ifnum\tn@named=1 \def\tn@pvn{#1}\tn@eachindex{#1}\tn@portnamed
  \else \def\tn@pnm{}\tn@nop{#1}{ports}{#1}\fi}
\def\tn@portnamed#1#2{\def\tn@pnm{#1}%
  \tn@portvisit{\tn@pvn}{#2}{\tn@get{n@\tn@pvn @ik@#1}}}
\def\tn@countport#1#2#3{\global\advance\tn@npt by 1
  \tn@ifused{#1}{#2}{#3}\ifnum\tn@isused=1 \global\advance\tn@ndr by 1 \fi}
\def\tn@drawn#1{\global\tn@ndr=0 \global\tn@npt=0
  \tn@ports{#1}\tn@countport
  \edef\tn@ndrawn{\the\tn@ndr}\edef\tn@nports{\the\tn@npt}}

% ---- types ------------------------------------------------------------------------
% \tntype{<name>}{<options>}
% A type of the figure's own: the style <name>, which is <options>, and which
% a tensor of it says it is -- \tnshow and an error name it. What a type is
% (tikz-tensors-nodes.tex) is said in its options: its outline, and its
% indices (tn index), named, which the layout then holds a figure to.
%   \tntype{site}{tn circle, tn index={l:left, r:right, s:down}}
\newcommand{\tntype}[2]{%
  \tikzset{#1/.style={#2, tn@tname={#1}}}\tn@props{#1}}
% how a tensor is named in a message: <tensor>, and (type <name>) if it has one
\def\tn@called#1{`#1'\ifcsname tn@n@#1@tname\endcsname
  \expandafter\ifx\csname tn@n@#1@tname\endcsname\@empty\else
  \space(type \tn@get{n@#1@tname})\fi\fi}

% When a picture ends, every index a type names (tn index) is drawn -- joined,
% or opened -- or the figure is an error naming the ones that are not: an
% index left out by accident is caught where it is made rather than in print.
% A picture inside a picture is checked with it.
\gdef\tn@ranked{}
\def\tn@pnm{}
\newcount\tn@depth
\def\tn@checkranks{%
  \foreach \tn@rn in \tn@ranked{%
    \expandafter\tn@checkrank\expandafter{\tn@rn}}%
  \gdef\tn@ranked{}}
\newcount\tn@nud
\def\tn@checkrank#1{%
  \gdef\tn@undrawn{}\global\tn@nud=0 \tn@ports{#1}\tn@undrawnport
  \ifx\tn@undrawn\@empty\else
    \PackageError{tikz-tensors}{\tn@called{#1}: \ifnum\tn@nud=1 its index
      \tn@undrawn\space is\else its indices \tn@undrawn\space are\fi\space
      not drawn}{Its type says tn index={\tn@get{n@#1@index}}, and a picture
      draws every index it names. \string\tnshow{#1} writes each of them to
      the log; \string\tnopen{rest}{#1} opens the free ones.}%
  \fi}
\def\tn@undrawnport#1#2#3{\tn@ifused{#1}{#2}{#3}%
  \ifnum\tn@isused=0 \global\advance\tn@nud by 1
    \ifx\tn@undrawn\@empty \xdef\tn@undrawn{\tn@pnm}%
    \else \xdef\tn@undrawn{\tn@undrawn, \tn@pnm}\fi\fi}
\tikzset{every picture/.append style={%
  execute at begin picture={\global\advance\tn@depth by 1 },
  execute at end picture={\global\advance\tn@depth by -1
    \ifnum\tn@depth=0 \tn@checkranks\fi}}}

% ---- lists of names -------------------------------------------------------------
% \tn@xlist{<list>}: the list, with every item written <pre>{<a>..<b>}<post>
% made the items <pre>a<post>, ..., <pre>b<post> -- S-ket-{2..4}, P-{1..4}-2
% -- as \tn@xl. Every command that takes a list of tensors or ports reads it
% through this.
\def\tn@xlist#1{%
  \gdef\tn@xl{}%
  \foreach \tn@xi in {#1}{\expandafter\tn@xone\tn@xi{\tn@nil}\relax\tn@end}}
\def\tn@xone#1#{\tn@xtwo{#1}}
\def\tn@xtwo#1#2#3\tn@end{%
  \def\tn@xr{#2}\def\tn@xnil{\tn@nil}%
  \ifx\tn@xr\tn@xnil \tn@xadd{#1}%
  \else
    \tn@xpost#3%
    \tn@xab#2\tn@end
    \foreach \tn@xk in {\tn@xa,...,\tn@xb}{\tn@xadd{#1\tn@xk\tn@xp}}%
  \fi}
\def\tn@xpost#1#{\def\tn@xp{#1}\@gobble}
\def\tn@xab#1..#2\tn@end{\edef\tn@xa{#1}\edef\tn@xb{#2}}
\def\tn@xadd#1{\ifx\tn@xl\@empty \xdef\tn@xl{#1}\else \xdef\tn@xl{\tn@xl,#1}\fi}

% ---- a tensor, shown ------------------------------------------------------------
% \tnshow{<tensor>, ...}
% What the package made of each tensor, written to the log (and the terminal):
% its block, the sites and layers it covers, its type, the sides its type
% gives it indices on, and each index -- its side, where on it, and whether it
% is drawn yet, joined or opened. Everything a layout decided without being
% told, said; for finding out why a figure looks as it does.
\newcommand{\tnshow}[1]{%
  \tn@xlist{#1}%
  \foreach \tn@sn in \tn@xl{\expandafter\tn@show\expandafter{\tn@sn}}}
\def\tn@show#1{%
  \tn@isnode{#1}{\string\tnshow}%
  \edef\tn@sb{\tn@get{n@#1@s}}\edef\tn@sk{\tn@get{\tn@sb @kind}}%
  \def\tn@skg{grid}\ifx\tn@sk\tn@skg \def\tn@skr{rows}\else \def\tn@skr{layers}\fi
  \typeout{tikz-tensors: #1 -- a tensor of \tn@sk\space `\tn@sb',
    sites \tn@get{n@#1@a}..\tn@get{n@#1@b}, \tn@skr\space
    \tn@get{n@#1@k}..\tn@get{n@#1@kb}}%
  \ifcsname tn@n@#1@type\endcsname
    \edef\tn@stn{\tn@get{n@#1@tname}}%
    \typeout{\space\space type: \ifx\tn@stn\@empty \tn@get{n@#1@type}\else
      \tn@stn\space (\string\tntype), drawn as \tn@get{n@#1@type}\fi}\fi
  \tn@drawn{#1}\edef\tn@srk{\tn@get{n@#1@rank}}%
  \typeout{\space\space rank: \ifx\tn@srk\@empty not said\else \tn@srk\fi
    ; \tn@ndrawn\space of \tn@nports\space indices drawn}%
  \ifcsname tn@n@#1@legs\endcsname \edef\tn@hl{\tn@get{n@#1@legs}}\else\def\tn@hl{}\fi
  \typeout{\space\space sides with indices: \ifx\tn@hl\@empty every side its
    layout gives it (tn legs not said)\else \tn@hl\fi}%
  \tn@ports{#1}\tn@showport}
% unless it is a triangle pointing up or down, which is only as tall as its
% middle near its apex -- then just inside its base.
\def\tn@inner#1{%
  \edef\tn@one{\tn@get{n@#1@one}}%
  \ifx\tn@one\tn@sup \tn@anc{#1}{south}\tn@len\tn@ay{\tn@ay+\tn@inset}%
  \else\ifx\tn@one\tn@sdown \tn@anc{#1}{north}\tn@len\tn@ay{\tn@ay-\tn@inset}%
  \else \tn@anc{#1}{center}\fi\fi}
% \tn@hport{<tensor>}{left|right}: its port on that side, as \tn@px (where the
% index starts) and \tn@pe (the x of the border); the height is the layer's.
\def\tn@hport#1#2{%
  \edef\tn@t{#2}%
  \ifx\tn@t\tn@sright
    \tn@anc{#1}{east}\let\tn@pe\tn@ax \tn@len\tn@px{\tn@ax-\tn@inset}%
  \else
    \tn@anc{#1}{west}\let\tn@pe\tn@ax \tn@len\tn@px{\tn@ax+\tn@inset}%
  \fi}
% \tn@vport{<tensor>}{up|down}: its port on that side in column \tn@cx, as
% \tn@px, \tn@py (where the index starts) and \tn@pe (the y of the border) --
% the layout's operation, since where a column meets a tensor is the layout's.
\def\tn@vport#1#2{\tn@nop{#1}{vport}{#1}{#2}}

% ---- what a figure writes --------------------------------------------------------
% \tnconnect[along=<style>, down=<style>, apart={<layer>, ...}]{<block>}
% Every index the layout implies, and nothing else; what that is, is each
% layout's to say (tikz-tensors-stack.tex, tikz-tensors-grid.tex). along=
% styles the indices along the layers (or the lattice), down= the ones down
% the sites, so that a canonical form's bonds carry arrows and its physical
% indices do not; both are tn edge unless said. apart= names layers whose
% tensors sit side by side without an index between them.
\pgfkeys{/tn/connect/.is family, /tn/connect/along/.initial=tn edge,
         /tn/connect/down/.initial=tn edge, /tn/connect/apart/.initial=}
\newcommand{\tnconnect}[2][]{%
  \pgfkeys{/tn/connect/.cd, along=tn edge, down=tn edge, apart=, #1}%
  \pgfkeysgetvalue{/tn/connect/apart}\tn@apart
  % apart= names layers, and ports: a port is one bond left out
  \expandafter\tn@xlist\expandafter{\tn@apart}\gdef\tn@apartl{}%
  \foreach \tn@ai in \tn@xl{%
    \expandafter\tn@apartone\tn@ai:\tn@end}%
  \let\tn@apart\tn@apartl
  \pgfkeysgetvalue{/tn/connect/along}\tn@along
  \pgfkeysgetvalue{/tn/connect/down}\tn@down
  \tn@op{#2}{connect}{#2}}

% \tnopen[edge=<style>, label=<text>, labels={<text>, ...}]{<direction>}{<item>, ...}
% Open indices toward <direction>. An item is a tensor -- its own indices that
% way -- or a block -- every free index of it that way. Where each ends, and
% the coordinate it leaves there, is the layout's to say. edge= is the edge
% type (tn edge unless said). label= puts a label at the end of each index,
% beyond it -- below one opened down, above one up, left and right of one
% opened sideways -- with #1 standing for its number among the indices this
% command opens, in order: label=$\sigma_{#1}$. labels= gives them one by one,
% in the same order, for numbering that is not 1, 2, 3: labels={$n_1$, $n_N$}.
\pgfkeys{/tn/open/.is family,
  /tn/open/edge/.initial=tn edge,
  /tn/open/label/.code={\gdef\tn@olab##1{#1}\gdef\tn@olmode{t}},
  /tn/open/labels/.code={\gdef\tn@olabs{#1}\gdef\tn@olmode{l}}}
\newcount\tn@ocount
\def\tn@olt{t}\def\tn@oll{l}
\def\tn@openkeys#1{%
  \gdef\tn@olmode{}\global\tn@ocount=0
  \pgfkeys{/tn/open/.cd, edge=tn edge, #1}%
  \pgfkeysgetvalue{/tn/open/edge}\tn@oedge}
% A layout calls this on the coordinate at the end of each index it opens.
% Toward a diagonal of a grid, the label goes beyond it on that diagonal.
\def\tn@olpos@nw{above left}\def\tn@olpos@ne{above right}
\def\tn@olpos@se{below right}\def\tn@olpos@sw{below left}
\def\tn@openlabel#1{%
  \global\advance\tn@ocount by 1
  \ifx\tn@dir\tn@sdown \def\tn@olpos{below}%
  \else\ifx\tn@dir\tn@sup \def\tn@olpos{above}%
  \else\ifcsname tn@olpos@\tn@dir\endcsname
    \edef\tn@olpos{\csname tn@olpos@\tn@dir\endcsname}%
  \else \let\tn@olpos\tn@dir \fi\fi\fi
  \ifx\tn@olmode\tn@olt
    \expandafter\tnput\expandafter[\tn@olpos]{#1}{\tn@olab{\the\tn@ocount}}%
  \else\ifx\tn@olmode\tn@oll
    \global\let\tn@lpick\@empty
    \foreach \tn@x [count=\tn@xc] in \tn@olabs{%
      \ifnum\tn@xc=\tn@ocount \global\let\tn@lpick\tn@x \fi}%
    \ifx\tn@lpick\@empty
      \PackageError{tikz-tensors}{\string\tnopen: more indices than labels}{}%
    \else \expandafter\tnput\expandafter[\tn@olpos]{#1}{\tn@lpick}\fi
  \fi\fi}
% \tnopen{rest}{<item>, ...} opens every index of each item not yet drawn,
% each toward its own side: on a tensor, all its free ones; on a block, those
% of each of its tensors whose type names its indices (tn index) -- a tensor
% that does not has as many as a layout gives it, which is more than it
% means; on a port (\tn@port), that index. An index left out by apart= is
% not opened. A named index opened ends at <tensor>-<name>.
\def\tn@srest{rest}
\newcommand{\tnopen}[3][]{%
  \tn@openkeys{#1}%
  \edef\tn@dir{#2}%
  \ifx\tn@dir\tn@srest \expandafter\tn@openrest \else \expandafter\tn@opendir \fi
  {#3}}
\def\tn@openrest#1{%
  \tn@xlist{#1}%
  \foreach \tn@it in \tn@xl{%
    \expandafter\tn@hascolon\tn@it:\tn@end
    \ifnum\tn@colon=1
      \expandafter\tn@port\expandafter{\tn@it}{\string\tnopen}%
      \tn@free{\tn@pn}{\tn@ps}{\tn@pkey}{\string\tnopen}%
      \tn@nop{\tn@pn}{openport}{\tn@oedge}{\tn@pn}{\tn@ps}{\tn@pkey}%
    \else\ifcsname tn@\tn@it @kind\endcsname
      \edef\tn@rns{\tn@get{\tn@it @nodes}}%
      \foreach \tn@rn in \tn@rns{%
        \edef\tn@rl{\tn@get{n@\tn@rn @legs}}%
        \ifx\tn@rl\@empty\else \expandafter\tn@restnode\expandafter{\tn@rn}\fi}%
    \else
      \tn@isnode{\tn@it}{\string\tnopen}%
      \expandafter\tn@restnode\expandafter{\tn@it}%
    \fi\fi}}
\def\tn@hascolon#1:#2\tn@end{\def\tn@cl{#2}%
  \ifx\tn@cl\@empty \def\tn@colon{0}\else \def\tn@colon{1}\fi}
\def\tn@restnode#1{\tn@ports{#1}\tn@restport}
\def\tn@restport#1#2#3{%
  \tn@ifused{#1}{#2}{#3}\tn@ifcut{#1}{#2}{#3}%
  \ifnum\tn@isused\tn@iscut=00 \tn@nop{#1}{openport}{\tn@oedge}{#1}{#2}{#3}\fi}
\def\tn@opendir#1{%
  \tn@xlist{#1}%
  \foreach \tn@it in \tn@xl{%
    \ifcsname tn@\tn@it @kind\endcsname
      \expandafter\tn@op\expandafter{\tn@it}{openblock}{\tn@oedge}{\tn@it}%
    \else
      \tn@isnode{\tn@it}{\string\tnopen}%
      \expandafter\tn@nop\expandafter{\tn@it}{opennode}{\tn@oedge}{\tn@it}%
    \fi}}

% \tnopenswap[edge=<style>, label=..., labels=...]{<tensor>}
% The two indices down from a tensor across two sites, exchanged on the way: a
% swap of two fermion legs. The keys are \tnopen's.
\newcommand{\tnopenswap}[2][]{%
  \tn@openkeys{#1}\def\tn@dir{down}%
  \tn@isnode{#2}{\string\tnopenswap}\tn@nop{#2}{openswap}{\tn@oedge}{#2}}

% \tnjoin[edge=<style>]{<port>}{<port>}
% One index between two ports the layout does not join: the bond that closes
% a periodic chain, a trace, a bond that skips over its neighbours. Both
% tensors are in one block. A port is <tensor>:<side>[:<n>] -- the <n>th
% index on that side, the first unless said -- or, of a tensor whose type
% names its indices, <tensor>:<name> or <tensor>:<n>, the <n>th of them
% (\tn@port). How it runs is the layout's to say. It is
% drawn from the first port to the second, which is the way an arrow on it
% points; a port that is drawn already is an error.
\def\tn@pspec#1:#2:#3:#4\tn@end{\def\tn@pn{#1}\def\tn@ps{#2}\def\tn@pk{#3}}
% one item of apart=: a layer, kept in \tn@apartl, or a port, marked cut
\def\tn@apartone#1:#2\tn@end{\def\tn@ap{#2}%
  \ifx\tn@ap\@empty
    \ifx\tn@apartl\@empty \xdef\tn@apartl{#1}\else \xdef\tn@apartl{\tn@apartl,#1}\fi
  \else
    \tn@port{#1:#2}{\string\tnconnect}%
    \tn@cut{\tn@pn}{\tn@ps}{\tn@pkey}%
  \fi}
\pgfkeys{/tn/join/.is family, /tn/join/edge/.initial=tn edge}
\newcommand{\tnjoin}[3][]{%
  \pgfkeys{/tn/join/.cd, edge=tn edge, #1}\pgfkeysgetvalue{/tn/join/edge}\tn@jedge
  \tn@port{#2}{\string\tnjoin}%
  \edef\tn@ja{\tn@get{n@\tn@pn @s}}%
  \edef\tn@jargs{{\tn@pn}{\tn@ps}{\tn@pkey}}%
  \tn@port{#3}{\string\tnjoin}%
  \edef\tn@jb{\tn@get{n@\tn@pn @s}}%
  \edef\tn@jargs{\tn@jargs{\tn@pn}{\tn@ps}{\tn@pkey}}%
  \ifx\tn@ja\tn@jb\else
    \PackageError{tikz-tensors}{\string\tnjoin: `#2' and `#3' are in two
      blocks}{An index is joined inside one stack, tree or grid.}\fi
  \expandafter\tn@joinop\expandafter{\tn@jargs}}
\def\tn@joinop#1{\tn@op{\tn@ja}{join}{\tn@jedge}#1}
