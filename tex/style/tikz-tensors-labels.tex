% tikz-tensors-labels.tex -- text on a diagram, placed by what it labels.

\tikzset{tn label/.style = {font=\small, inner sep=2pt}}

% \tnput[<placement>]{<point>}{<text>}      text on or beside one point
% \tnmid[<placement>]{<a>}{<b>}{<text>}     text midway between two points
% Where a label goes is said in terms of what it labels -- above this tensor,
% below that index's end, midway along this bond -- and never as a coordinate,
% because a coordinate written for a label is a coordinate that has to be
% re-found every time the thing it labels moves. <placement> is TikZ's: above,
% below, left, right, center, and any of them with a distance.
% A label counts as drawn when the next equals sign is placed.
\newcommand{\tnput}[3][above]{\node[tn label, #1] (tn@lab) at (#2) {#3};\tn@tracenode{tn@lab}%
  \tn@anc{tn@lab}{east}\tn@reach{\tn@ax}}
\newcommand{\tnmid}[4][above]{\node[tn label, #1] (tn@lab) at ($(#2)!0.5!(#3)$) {#4};\tn@tracenode{tn@lab}%
  \tn@anc{tn@lab}{east}\tn@reach{\tn@ax}}

% \tnframe[label=<text>]{<tensor>, ...}
% A dashed box around the named tensors, framesep clear of them, with <text>
% above its right corner: the tensors a step contracts into one, or a part of
% a network set apart. Said by what it frames and never by a coordinate, like
% a label. It counts as drawn when the next equals sign is placed.
\pgfkeys{/tn/frame/.is family, /tn/frame/label/.initial=}
% Its measures are \tn@fm...: \tn@fx, \tn@fxa and the like are the way an
% index runs (\tn@fluxof, \tn@orient in tikz-tensors-layout.tex), which a
% \tnjoin before the frame has just defined.
\newdimen\tn@fmxa \newdimen\tn@fmya \newdimen\tn@fmxb \newdimen\tn@fmyb
\newcommand{\tnframe}[2][]{%
  \pgfkeys{/tn/frame/.cd, label=, #1}\pgfkeysgetvalue{/tn/frame/label}\tn@flab
  \global\tn@fmxa=\maxdimen \global\tn@fmya=\maxdimen
  \global\tn@fmxb=-\maxdimen \global\tn@fmyb=-\maxdimen
  \tn@xlist{#2}%
  \foreach \tn@fn in \tn@xl{%
    \tn@anc{\tn@fn}{south west}%
    \ifdim\tn@ax<\tn@fmxa \global\tn@fmxa=\tn@ax\fi
    \ifdim\tn@ay<\tn@fmya \global\tn@fmya=\tn@ay\fi
    \tn@anc{\tn@fn}{north east}%
    \ifdim\tn@ax>\tn@fmxb \global\tn@fmxb=\tn@ax\fi
    \ifdim\tn@ay>\tn@fmyb \global\tn@fmyb=\tn@ay\fi}%
  \tn@len\tn@fmw{\the\tn@fmxb-\the\tn@fmxa+2*\tn@framesep}%
  \tn@len\tn@fmh{\the\tn@fmyb-\the\tn@fmya+2*\tn@framesep}%
  \tn@len\tn@fmx{(\the\tn@fmxa+\the\tn@fmxb)/2}\tn@len\tn@fmy{(\the\tn@fmya+\the\tn@fmyb)/2}%
  \node[tn frame, rectangle, inner sep=0pt, minimum width=\tn@fmw, minimum height=\tn@fmh]
    (tn@frame) at (\tn@fmx,\tn@fmy) {};\tn@tracenode{tn@frame}%
  \tn@anc{tn@frame}{east}\tn@reach{\tn@ax}%
  \ifx\tn@flab\@empty\else
    \expandafter\tnput\expandafter[above right]{tn@frame.north east}{\tn@flab}\fi}
